import jakarta.servlet.*;
import jakarta.servlet.http.*;
import java.io.*;
import java.util.*;

public class TransactionServlet extends HttpServlet {
    private static final String TRANSACTION_HISTORY_ATTRIBUTE = "transactionHistory";
    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
        response.setContentType("text/html");
        PrintWriter out = response.getWriter();
        // Get or create session
        HttpSession session = request.getSession(true);
        List<String> transactionHistory = (List<String>) session.getAttribute(TRANSACTION_HISTORY_ATTRIBUTE);
        // Initialize transaction history if it doesn't exist
        if (transactionHistory == null) {
            transactionHistory = new ArrayList<>();
            session.setAttribute(TRANSACTION_HISTORY_ATTRIBUTE, transactionHistory);
        }
        // Get item details from request parameters
        String itemName = request.getParameter("name");
        String itemId = request.getParameter("id");

        // Validate and add transaction
        if (itemName != null && !itemName.trim().isEmpty() && itemId != null && !itemId.trim().isEmpty()) {
            String newTransaction = "Purchased Item: " + itemName + " (ID: " + itemId + ")";
            transactionHistory.add(newTransaction);
        }
        // Display transaction history
        out.println("<html><body>");
        out.println("<h2>User Transaction History</h2>");
        out.println("<p>Current Session ID: " + session.getId() + "</p>");

        out.println("<h3>Your Transaction History:</h3>");
        if (transactionHistory.isEmpty()) {
            out.println("<p>No transactions made yet.</p>");
        } else {
            out.println("<ul>");
            for (String transaction : transactionHistory) {
                out.println("<li>" + transaction + "</li>");
            }
            out.println("</ul>");
        }
        // Navigation links
        out.println("<a href='/ViewProductsServlet/fetchData'>View Available Products</a><br>");
        out.println("<a href='/TransactionServlet'>Add Another Transaction</a>");
        out.println("</body></html>");
    }
}
